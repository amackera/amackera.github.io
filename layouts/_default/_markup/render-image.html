{{/* Site override of themes/typo/layouts/_default/_markup/render-image.html. Differences from the theme: - Raster images that are page or site resources are converted to WebP and capped at 1600px wide (2x the 780px column), which cuts most screenshot pages by 70-80%. SVGs and GIFs are served untouched. - .Width/.Height are only read on raster resources, so SVG page resources render instead of failing the build. Must stay compatible with the Hugo version in .github/workflows/hugo.yaml. */}} {{- $u := urls.Parse .Destination -}} {{- $url := $u.String -}} {{- $img := "" -}} {{- if not $u.IsAbs -}} {{- $path := strings.TrimPrefix "./" $u.Path -}} {{- with .PageInner.Resources.Get $path }}{{ $img = . }}{{ end -}} {{- if not $img }}{{ with .PageInner.Parent }}{{ with .Resources.Get $path }}{{ $img = . }}{{ end }}{{ end }}{{ end -}} {{- if not $img }}{{ with resources.Get $path }}{{ $img = . }}{{ end }}{{ end -}} {{- with $img -}} {{- $raster := and (eq .ResourceType "image") (ne .MediaType.SubType "svg") (ne .MediaType.SubType "gif") -}} {{- if $raster -}} {{- if gt .Width 1600 -}} {{- $img = .Resize "1600x webp q82" -}} {{- else -}} {{- $img = .Process "webp q82" -}} {{- end -}} {{- end -}} {{- $url = $img.RelPermalink -}} {{- with $u.RawQuery }}{{ $url = printf "%s?%s" $url . }}{{ end -}} {{- with $u.Fragment }}{{ $url = printf "%s#%s" $url . }}{{ end -}} {{- end -}} {{- end -}} {{/* Everything after "#" in the destination is a space of style tags, e.g. image.png#full -> class "img-full" */}} {{ $url = $url | safeURL }} {{ $file_name_array := split $url "#" }} {{ $classes := "" }} {{ range $idx := seq (sub (len $file_name_array) 1) }} {{ $tag := index $file_name_array $idx }} {{ $classes = printf "%s img-%s" $classes $tag}} {{ end }} {{ $sized := false }} {{ with $img }}{{ $sized = and (eq .ResourceType "image") (ne .MediaType.SubType "svg") }}{{ end }}
